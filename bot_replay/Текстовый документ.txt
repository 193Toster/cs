using CounterStrikeSharp.API;
using CounterStrikeSharp.API.Core;
using CounterStrikeSharp.API.Modules.Commands;
using CounterStrikeSharp.API.Modules.Utils;
using System.Numerics;

namespace ActionRecorder;

public class Plugin : BasePlugin
{
    public override string ModuleName => "Recorder";
    public override string ModuleVersion => "1.0";

    private bool recording;
    private CCSPlayerController? recordPlayer;
    private List<(Vector3 pos, QAngle ang)> frames = new();
    private CounterStrikeSharp.API.Modules.Timers.Timer? timer;

    public override void Load(bool hotReload)
    {
        AddCommand("css_record", "Start", (p, _) => {
            if (p == null || recording) return;
            recording = true;
            recordPlayer = p;
            frames.Clear();
            timer = AddTimer(0.0f, () => {
                if (!recording || recordPlayer?.PlayerPawn.Value == null) { StopRec(); return; }
                var pawn = recordPlayer.PlayerPawn.Value;
                frames.Add((pawn.AbsOrigin!, pawn.EyeAngles!));
            }, CounterStrikeSharp.API.Modules.Timers.TimerFlags.REPEAT);
            p.PrintToChat($" {ChatColors.Green}Recording started");
        });

        AddCommand("css_stop", "Stop", (p, _) => {
            if (p == null || !recording) return;
            StopRec();
            p.PrintToChat($" {ChatColors.Green}Stopped ({frames.Count} frames)");
        });

        AddCommand("css_replay", "Replay", (p, _) => {
            if (p == null || frames.Count == 0) return;
            var bot = Utilities.CreateEntityByName<CCSPlayerController>("cs_player_controller");
            if (bot == null) return;
            bot.PlayerName = "ReplayBot";
            bot.TeamNum = (byte)CsTeam.Terrorist;
            bot.IsBot = true;
            bot.DispatchSpawn();
            _ = Replay(bot);
        });
    }

    private void StopRec() { recording = false; recordPlayer = null; timer?.Kill(); timer = null; }

    private async Task Replay(CCSPlayerController bot)
    {
        var pawn = bot.PlayerPawn.Value;
        if (pawn == null) return;
        var delay = (int)(Server.TickInterval * 1000);
        foreach (var (pos, ang) in frames)
        {
            if (pawn == null || !pawn.IsValid) break;
            pawn.Teleport(pos, ang, Vector3.Zero);
            await Task.Delay(delay);
        }
        Server.NextFrame(() => { pawn?.Remove(); bot.Remove(); });
    }
}
